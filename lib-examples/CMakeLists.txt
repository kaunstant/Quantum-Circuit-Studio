cmake_minimum_required(VERSION 3.20)
project(QCSLibraryExamples LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(QCS CONFIG REQUIRED)

file(GLOB EXAMPLE_PROGRAMS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
foreach(EXAMPLE_SOURCE IN LISTS EXAMPLE_PROGRAMS)
    get_filename_component(EXAMPLE_NAME "${EXAMPLE_SOURCE}" NAME_WE)
    add_executable("${EXAMPLE_NAME}" "${EXAMPLE_SOURCE}")
    target_link_libraries("${EXAMPLE_NAME}" PRIVATE QCS::QCS)
endforeach()
